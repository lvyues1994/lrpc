set(lrpc_top_level OFF)
if(CMAKE_CURRENT_SOURCE_DIR STREQUAL CMAKE_SOURCE_DIR)
    set(lrpc_top_level ON)
endif()
option(LRPC_BUILD_TESTS "Build lrpc tests" ${lrpc_top_level})
option(LRPC_BUILD_ALLOCATION_TESTS "Build the allocation gate test, which replaces global operator new (not under TSan)" ON)
option(LRPC_BUILD_RPC "Build the RPC library, which needs net" ON)
option(LRPC_BUILD_PROTOBUF "Build the optional protobuf codec" OFF)
option(LRPC_BUILD_JSON "Build the optional typed JSON codec" OFF)
option(LRPC_BUILD_CODEGEN "Build protoc-gen-rpc" ${LRPC_BUILD_PROTOBUF})
option(LRPC_BUILD_EXAMPLES "Build the examples" ${lrpc_top_level})
option(LRPC_BUILD_BENCHMARKS "Build the single-shard TCP latency benchmark" OFF)
option(LRPC_BENCH_COUNT_ALLOCATIONS "Instrument benchmark new/new[] calls (separate from latency runs)" OFF)
option(LRPC_ENABLE_COMPRESSION "Enable bounded zstd-frame and LZ4-block message compression" OFF)
option(LRPC_ENABLE_SANITIZERS "Enable AddressSanitizer and UBSan" OFF)
option(LRPC_BUILD_FUZZER "Build the Clang libFuzzer wire harness" OFF)

function(lrpc_configure_target target)
    target_compile_features(${target} PUBLIC cxx_std_14)
    set_target_properties(${target} PROPERTIES
        CXX_STANDARD 14 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${target} PRIVATE
            -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion
            -Werror=return-type -Werror=uninitialized -Werror=vla
            -Wnon-virtual-dtor -Wold-style-cast)
        if(LRPC_ENABLE_SANITIZERS)
            target_compile_options(${target} PUBLIC
                -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(${target} PUBLIC -fsanitize=address,undefined)
        endif()
    elseif(LRPC_ENABLE_SANITIZERS)
        message(FATAL_ERROR "LRPC_ENABLE_SANITIZERS requires GCC or Clang")
    endif()
endfunction()

add_subdirectory(wire)
add_subdirectory(compression)
if(LRPC_BUILD_RPC)
    include(cmake/NetDependency.cmake)
    add_library(lrpc STATIC
        src/core.cpp src/engine.cpp src/stream_core.cpp src/client.cpp src/client_stream.cpp
        src/channel.cpp src/server.cpp src/transport.cpp)
    add_library(lrpc::rpc ALIAS lrpc)
    target_include_directories(lrpc PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
    target_link_libraries(lrpc PUBLIC lrpc::wire net::net PRIVATE lrpc::compression)
    lrpc_configure_target(lrpc)
endif()
